параметры-ссылки вроде self.

```python
class Russian:
    @staticmethod
    def greeting():
        print("Привет")

    def greeting():
        print("Hello")

    language.greeting()
```

```python
# Класс Tomato
class Tomato:
    # Статическое свойство states, которое хранит стадии созревания помидора
    states = {0: "отсутствует", 1: "цветение", 2: "зеленый", 3: "красный"}
    
    def __init__(self, index):
        # Динамическое свойство _index: уникальный идентификатор для каждого томата
        # Передаётся параметром при создании объекта Tomato
        self._index = index
        # Динамическое свойство _state: хранит текущую стадию созревания томата
        # Изначально присваивается 0, что соответствует стадии "отсутствует" из словаря states
        self._state = 0  
    
    def grow(self):
        """Метод для перевода томата на следующую стадию созревания"""
        # Проверяем, достиг ли томат последней стадии созревания
        if self._state < 3:
            # Если нет, то увеличиваем индекс _state на 1, что переводит на следующую стадию
            self._state += 1  
    
    def is_ripe(self):
        """Метод для проверки, созрел ли томат (находится ли на последней стадии)"""
        # Возвращаем True, если _state равен 3 (стадия "красный"), иначе False
        return self._state == 3  
    
    def get_state(self):
        """Метод для получения текущей стадии созревания в виде строки"""
        # Используем словарь states для преобразования числового состояния _state в текстовое описание
        return Tomato.states[self._state]


# Класс TomatoBush
class TomatoBush:
    def __init__(self, num_tomatoes):
        # Динамическое свойство tomatoes: список объектов Tomato
        # Создаётся путём генерации заданного количества объектов Tomato с уникальными индексами
        self.tomatoes = [Tomato(index) for index in range(num_tomatoes)]
    
    def grow_all(self):
        """Метод для перевода всех томатов на кусте на следующую стадию созревания"""
        # Проходим по каждому томату в списке tomatoes и вызываем метод grow для обновления стадии
        for tomato in self.tomatoes:
            tomato.grow()
    
    def all_are_ripe(self):
        """Метод, проверяющий, созрели ли все томаты на кусте"""
        # Используем встроенную функцию all(), чтобы убедиться, что все томаты созрели
        return all(tomato.is_ripe() for tomato in self.tomatoes)
    
    def give_away_all(self):
        """Метод для сбора урожая, который очищает список томатов"""
        # Очищаем список tomatoes, что символизирует сбор урожая
        self.tomatoes = []


# Класс Gardener
class Gardener:
    def __init__(self, name, plant):
        # Публичное свойство name: хранит имя садовника, передаётся параметром
        self.name = name
        # Динамическое свойство _plant: хранит объект TomatoBush, который садовник будет выращивать
        self._plant = plant
    
    def work(self):
        """Метод, который заставляет садовника ухаживать за растением"""
        # Выводим сообщение о том, что садовник ухаживает за растением
        print(f"{self.name} ухаживает за растением...")
        # Вызываем метод grow_all() у растения (куста), чтобы все томаты стали более зрелыми
        self._plant.grow_all()
    
    def harvest(self):
        """Метод, который проверяет, созрели ли все плоды, и собирает урожай"""
        # Если все томаты созрели, садовник собирает урожай, очищая куст
        if self._plant.all_are_ripe():
            print(f"{self.name} собирает урожай.")
            self._plant.give_away_all()
        else:
            # Если томаты еще не созрели, выводим предупреждение
            print("Помидоры еще не созрели, нужно подождать.")
    
    @staticmethod
    def knowledge_base():
        """Статический метод, который выводит справочную информацию по садоводству"""
        # Выводим информацию по садоводству для всех садоводов
        print("Справка по садоводству: Ухаживайте за растением, чтобы помидоры созревали. Собирайте урожай, когда все томаты красные.")


# Тестирование программы

# 1) Вызов справки по садоводству
# Статический метод, вызывается через имя класса Gardener
Gardener.knowledge_base()

# 2) Создаем объекты классов TomatoBush и Gardener
# Создаем куст с тремя томатами
bush = TomatoBush(3)
# Создаем садовника, который ухаживает за кустом
gardener = Gardener("Петр", bush)

# 3) Ухаживаем за кустом с помидорами
# Вызываем метод work() садовника, который заставляет куст расти
gardener.work()
# Проверяем состояние каждого томата на кусте после ухода
print("Состояние томатов после первого ухода:")
for tomato in bush.tomatoes:
    # Выводим текущую стадию созревания каждого томата
    print(tomato.get_state())

# 4) Пытаемся собрать урожай, когда томаты еще не дозрели
# Вызываем метод harvest() садовника, который проверяет зрелость томатов
gardener.harvest()

# Продолжаем ухаживать за томатами до полной зрелости
gardener.work()
gardener.work()  # Ухаживаем еще дважды, чтобы все томаты созрели
print("Состояние томатов после дополнительных уходов:")
for tomato in bush.tomatoes:
    # Выводим текущую стадию созревания каждого томата, чтобы убедиться, что все созрели
    print(tomato.get_state())

# 5) Собираем урожай
# После ухода все томаты должны быть зрелыми, вызываем harvest() для сбора урожая
gardener.harvest()
```
